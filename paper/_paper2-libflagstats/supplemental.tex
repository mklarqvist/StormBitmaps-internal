\begin{minted}[escapeinside=\\$\\$,mathescape=true]{c}
void mask_select_propagate_carry(__m512i* r, 
  __m512i select, __m512i compare,
  __m512i false_mask) 
{
  // *r & select == compare
  __mmask32 mask = _mm512_cmpeq_epi16_mask(*r & select, compare);
  // *r &= (m, m, m, $\ldots$, m), where m is in {false_mask, 65535}.
  *r &= (_mm512_maskz_set1_epi16(mask, 65535) | false_mask);
}
\end{minted}

algorithm for a single update for a \texttt{FLAG} word.}

\begin{minted}[escapeinside=\\$\\$,mathescape=true]{c}
void mask_select_propagate_carry2(__m512i* r, 
  __m512i select, __m512i compare, 
  __m512i false_mask)
{
  // *r & select == compare
  __mmask32 mask = _mm512_cmpeq_epi16_mask(*r & select, compare);
  int bin_fun = 0xE0; // a & (b | c) = 11100000b
  *r = _mm512_ternarylogic_epi64(*r, _mm512_maskz_set1_epi16(mask, 65535), 
         false_mask, bin_fun);
}
\end{minted}
